A try–except vezérlési útja

A kivétel megváltoztatja, mely utasítások futnak le.

A try blokkban megkíséreljük az adott műveletet. Ha a megnevezett kivétel fellép, a try további utasításai kimaradnak, és az illeszkedő except ág fut. A hibás művelethez a program nem tér vissza automatikusan; az újrapróbálkozáshoz külön ciklus kell.

def mereseket_beolvas(fajlnev):
    meresek = []
    hibak = []
    ures_sorok = 0
    sorszam = 0
    with open(fajlnev, "r", encoding="utf-8") as fajl:
        for sor in fajl:
            sorszam += 1
            tisztitott = sor.strip()
            if tisztitott == "":
                ures_sorok += 1
                continue
            try:
                ertek = int(tisztitott)
            except ValueError:
                hibak.append((sorszam, tisztitott))
                continue
            meresek.append(ertek)
    return meresek, hibak, ures_sorok


def mereseket_osszesit(ertekek):
    osszeg = 0
    maximum = None
    for ertek in ertekek:
        osszeg += ertek
        if maximum is None or ertek > maximum:
            maximum = ertek
    darab = len(ertekek)
    atlag = None
    if darab > 0:
        atlag = osszeg / darab
    return {"darab": darab, "osszeg": osszeg, "maximum": maximum, "atlag": atlag}


def jelentest_ment(fajlnev, adatok, hibak, ures_sorok):

            fajl.write("Maximum és átlag: nincs adat\n")
            fajl.write("Állapot: nincs feldolgozható mérés\n")
